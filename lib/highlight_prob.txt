// For Verse Numbers - replace the verse number section:
allSpans.add(TextSpan(
  text: verseNumText,
  style: TextStyle(
    fontSize: widget.fontSize * 0.8,
    fontWeight: FontWeight.bold,
    color: themeProvider.verseNumberColor,
    fontFamily: isArabic ? widget.fontFamily : 'serif',
    letterSpacing: 0.5,
    background: verseNumHighlighted 
        ? (Paint()
            ..color = Colors.yellow
            ..strokeWidth = widget.fontSize * 1.2  // Controls total vertical padding
            ..style = PaintingStyle.stroke
            ..strokeJoin = StrokeJoin.round)  // Rounds the corners
        : null,
    decoration: verseNumUnderlined ? TextDecoration.underline : null,
    decorationColor: Theme.of(context).primaryColor,
    decorationThickness: 2,
  ),
));

// For Footnote Numbers - replace the footnote section:
spans.add(
  TextSpan(
    text: '\u2066$footnoteNum\u2069',
    style: TextStyle(
      fontSize: widget.fontSize * 0.65,
      color: themeProvider.footnoteNumberColor,
      fontWeight: FontWeight.bold,
      fontFeatures: const [FontFeature.superscripts()],
      background: footnoteHighlighted
          ? (Paint()
              ..color = Colors.yellow
              ..strokeWidth = widget.fontSize * 0.8  // Controls vertical padding
              ..style = PaintingStyle.stroke
              ..strokeJoin = StrokeJoin.round)
          : null,
      decoration: footnoteUnderlined ? TextDecoration.underline : null,
      decorationColor: Theme.of(context).primaryColor,
      decorationThickness: 2,
    ),
    recognizer: TapGestureRecognizer()..onTap = () => _scrollToFootnote(footnoteNum),
  ),
);
